# Union of dependencies for every module in this reproducibility package.
# Prefer this file for a single environment that can run everything; each
# module also has its own narrower requirements where relevant
# (bert_models/requirements.txt).
#
# Install with: pip install -r requirements.txt
# (PyTorch Geometric may need a separate install step -- see note below.)
#
# PINNING POLICY: every package below that this codebase actually imports
# (i.e. everything except the two optional packages at the bottom) is
# pinned to the exact version it was most recently verified against end to
# end in this package's test environment (every module's smoke test /
# --no-wandb training run, the full gnn/ + explainability/ +
# risk_stratification/ pipeline, and baselines/train_rf_xgb.py). A loose
# `>=`-only range previously let an unrelated `pip install -U` silently
# bump a transitive dependency to a version that changed behavior (this is
# how the torch/transformers incompatibility documented below was first
# discovered). If you need a newer version of something, that's fine --
# just re-run each module's smoke test after upgrading and update the pin.

# --- Core data science ---
pandas==2.3.3
numpy==1.26.4            # keep <2.0.0: several modules' GNN edge-attribute
                          # construction and bert_models' tokenizer glue
                          # code were only verified against NumPy 1.x
scikit-learn==1.7.2
joblib==1.6.0
matplotlib==3.10.8

# --- Baselines (baselines/) ---
xgboost==3.2.0
shap==0.49.1
scikit-optimize==0.10.2  # optional: enables real BayesSearchCV; falls back to
                          # RandomizedSearchCV automatically if unavailable

# --- GNN + explainability + risk stratification (gnn/, explainability/, risk_stratification/) ---
# IMPORTANT: keep torch <2.5.0. This package was developed/tested against
# torch==2.1.2 (aarch64 / disk-space constraints -- see gnn/train.py's module
# docstring); torch_geometric==2.8.0.post1 works fine against it, but see the
# transformers note below for why the *upper* bound on torch also matters
# for bert_models/.
torch==2.1.2
torch_geometric==2.8.0.post1
tqdm==4.70.1

# --- BERT-family models (bert_models/) ---
# See bert_models/requirements.txt for the full, empirically-verified
# explanation. Short version: transformers>=4.41 (and all of transformers
# 5.x) fail to import or load checkpoints against torch<2.5 in this
# package's tested environment. transformers==4.40.0 is the newest release
# confirmed to work against torch==2.1.2.
transformers==4.40.0

# --- Optional: experiment tracking (all modules support --no-wandb / a
# local-checkpoint fallback that skips this entirely). Not exercised by this
# package's own test environment, so left as a minimum-version range rather
# than an exact pin -- pin it yourself once you've verified a version
# against your WandB account/server. ---
wandb>=0.15.0

# --- Optional: development/testing. Same rationale as wandb above. ---
pytest>=7.0.0
