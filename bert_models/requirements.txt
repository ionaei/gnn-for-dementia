# BERT-family models for dementia risk prediction
# Install with: pip install -r requirements.txt
#
# These are the same exact pins as the repository-root requirements.txt
# (see the PINNING POLICY note there) for the packages this directory
# actually imports -- kept here too so bert_models/ can be installed
# standalone without pulling in the GNN/baselines-only dependencies.

# Core ML/data science
torch==2.1.2
# IMPORTANT (verified empirically in this package's dev environment, which
# is pinned to torch==2.1.2 for aarch64/disk-space reasons -- see
# gnn/train.py's module docstring): transformers>=4.41 and all of
# transformers 5.x require internal `torch.utils._pytree` APIs that don't
# exist in torch 2.1.2 (AttributeError at import time), and transformers>=5.0
# additionally refuses to enable its PyTorch backend at all unless
# torch>=2.5 is installed (AutoModel*.from_pretrained(...) then raises
# "requires the PyTorch library but it was not found in your environment"
# even though torch IS installed). transformers==4.40.0 is the newest
# release confirmed to import and load checkpoints cleanly against
# torch==2.1.2 in this environment. If you upgrade torch to >=2.5, the
# transformers version ceiling below can be lifted too.
transformers==4.40.0
scikit-learn==1.7.2
pandas==2.3.3
numpy==1.26.4

# Optional: experiment tracking. Not exercised by this package's own test
# environment, so left as a minimum-version range rather than an exact pin.
wandb>=0.15.0

# Optional: development/testing. Same rationale as wandb above.
pytest>=7.0.0
